a = {'a': 1, 'b': 2, 'c': 3}
for x in a:
    print(x) 
    # a b c
for x in a.values():
    print(x) 
    # 1 2 3
for x in a.items():
    print(x) 
    # ('a', 1) ('b', 2) ('c', 3)
a = {'a': 'aa', 'b': 'bb', 'c': 'cc'}
for name, contents in a.items():
    if 'a' in contents:
        print(name) 
        # a
        print(contents) 
        # aa
a = {'a': 'aa', 'b': 'ab', 'c': 'bb'}
for name, contents in a.items():
    if 'a' in contents and not ('b' in contents or 'd'
        in contents):
        print(name) 
        # a
a = {'a': {'aa', 'ab'}, 'b': {'ab', 'bb'}}
for name, contents in a.items():
    if contents & { 'aa' }:
        print(name) 
        # a
